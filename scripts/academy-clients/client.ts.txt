function jwtRole(token?: string): string | undefined {
  try {
    if (!token || !/^[\w-]+\.[\w-]+\.[\w-]+$/.test(token)) return;
    return JSON.parse(atob(token.split(".")[1].replace(/-/g, "+").replace(/_/g, "/"))).role;
  } catch {
    return undefined;
  }
}

function requestHeaders(required: readonly string[], values: Credentials) {
  const headers = new Headers();
  for (const type of required) {
    switch (type) {
      case "apiKey":
        if (
          !values.apiKey ||
          (!/^sb_publishable_[\w-]+$/.test(values.apiKey) && jwtRole(values.apiKey) !== "anon")
        ) throw new Error("A publishable or legacy anon Academy key is required.");
        headers.set("apikey", values.apiKey);
        break;
      case "bearerAuth":
        if (!values.accessToken || jwtRole(values.accessToken) !== "authenticated")
          throw new Error("An Academy account access token is required.");
        headers.set("Authorization", "Bearer " + values.accessToken);
        break;
      case "integrationKey":
        if (!values.integrationKey || !/^lvi_[0-9a-f-]{36}_[0-9a-f]{64}$/.test(values.integrationKey))
          throw new Error("An Academy integration key is required.");
        headers.set("Authorization", "Bearer " + values.integrationKey);
        break;
    }
  }
  return headers;
}
